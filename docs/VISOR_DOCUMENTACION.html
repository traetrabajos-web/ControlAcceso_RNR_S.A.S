💻 Script SQL DDL y DML Completo (`database/schema.sql`)

📥 Descargar schema.sql

Contiene la definición completa de base de datos, 20 tablas relacionales, 3 vistas operativas, 3 triggers automáticos y todos los datos semilla de Recuperaciones Naranjo:

-- ============================================================================
-- BASE DE DATOS: SENTINELPRO ACCESS & IAM (RECUPERACIONES NARANJO S.A.S.)
-- SEDE: Zona Industrial Mamonal, Cartagena de Indias, Bolívar
-- ============================================================================

CREATE DATABASE IF NOT EXISTS `sentinelpro_iam`
CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE `sentinelpro_iam`;

-- 1. TABLA CONFIGURACION
CREATE TABLE `configuracion_empresa` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nombre_empresa` VARCHAR(150) NOT NULL DEFAULT 'Recuperaciones Naranjo Recycling S.A.S.',
    `nit_ruc` VARCHAR(30) NOT NULL DEFAULT '901.884.219-5',
    `jornada_semanal_horas` DECIMAL(4,1) NOT NULL DEFAULT 42.0,
    `tolerancia_tardanza_min` INT NOT NULL DEFAULT 10,
    `aforo_maximo_planta` INT NOT NULL DEFAULT 150
);

-- 2. TABLA TURNOS LABORALES (42 HORAS)
CREATE TABLE `turnos_laborales` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dia` VARCHAR(25) NOT NULL,
    `codigo_dia` INT NOT NULL,
    `hora_entrada` TIME NOT NULL,
    `hora_salida` TIME NOT NULL,
    `horas_ordinarias` DECIMAL(4,2) NOT NULL,
    `es_jornada_extra` BOOLEAN NOT NULL DEFAULT FALSE
);

-- 3. TABLA TRABAJADORES (PADRÓN MAESTRO NARANJO)
CREATE TABLE `trabajadores` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `codigo_credencial` VARCHAR(30) NOT NULL UNIQUE,
    `documento` VARCHAR(30) NOT NULL UNIQUE, -- Cédula de Ciudadanía (CC)
    `nombre_completo` VARCHAR(160) NOT NULL,
    `cargo_id` INT NOT NULL,
    `tipo_personal` ENUM('PATIO', 'ADMINISTRACION', 'APRENDIZ_SENA') NOT NULL,
    `es_puesto_critico` BOOLEAN NOT NULL DEFAULT FALSE,
    `arl_vigente` BOOLEAN NOT NULL DEFAULT TRUE,
    `estado` ENUM('ACTIVO', 'SUSPENDIDO', 'INACTIVO') NOT NULL DEFAULT 'ACTIVO',
    `ubicacion_actual` ENUM('DENTRO_PLANTA', 'FUERA_PLANTA') NOT NULL DEFAULT 'FUERA_PLANTA',
    FOREIGN KEY (`cargo_id`) REFERENCES `cargos`(`id`) ON DELETE RESTRICT
);
